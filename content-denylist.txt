# ── Telefoonnummers ──────────────────────────────────────────
# Belgisch gsm-nummer: 0470 12 34 56, 0470/12.34.56, +32 470 12 34 56, 0032 470 123456
re:(?<![\d+])(?:\+32|0032|0)[\s./-]?4\d{2}(?:[\s./-]?\d{2}){3}(?!\d)
# Elk nummer in internationaal formaat: + en minstens 8 cijfers
re:(?<![\w+])\+\d{1,3}(?:[\s./-]?\d){7,}(?!\d)
